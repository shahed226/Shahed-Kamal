2. Prime Number

import java.util.Scanner;

public class PrimeNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in); // user er theke input nibo tai
        System.out.print("Enter a number: ");
        int num = sc.nextInt();
        boolean isPrime = true; // shurute dhore nilam prime, pore lagle change korbo

        // 0 ar 1 prime na, tai age e bad dilam
        if (num <= 1) {
            isPrime = false;
        } else {
            // square root porjonto check korlei hoy, puro ta korar dorkar nai
            for (int i = 2; i <= Math.sqrt(num); i++) {
                if (num % i == 0) { // bhag gele mane prime na, loop thamiye dilam
                    isPrime = false;
                    break;
                }
            }
        }

        // result dekhacchi
        if (isPrime) {
            System.out.println(num + " is a Prime Number");
        } else {
            System.out.println(num + " is not a Prime Number");
        }
        sc.close();
    }
}


3. Palindrome Number

import java.util.Scanner;

public class PalindromeNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int num = sc.nextInt();
        int original = num; // ashol ta rekhe dilam, karon num to loop e change hoye jabe
        int reverse = 0;

        // ekta ekta digit niye number ta ulta cchi
        while (num > 0) {
            int digit = num % 10; // shesh er digit ta ber korlam
            reverse = reverse * 10 + digit; // ulta number er pichone boshiye dilam
            num = num / 10; // shesh er digit kete fellam
        }

        // ulta korleo jodi same thake tahole palindrome
        if (original == reverse) {
            System.out.println(original + " is a Palindrome Number");
        } else {
            System.out.println(original + " is not a Palindrome Number");
        }
        sc.close();
    }
}


4. Armstrong Number

import java.util.Scanner;

public class ArmstrongNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int num = sc.nextInt();
        int original = num; // shesh e milanor jonno ashol number ta rekhe dilam
        int sum = 0;
        int digits = String.valueOf(num).length(); // number ta te koyta digit ache gune nilam

        // prottek digit er power (digit shonkha) niye jog korchi
        while (num > 0) {
            int digit = num % 10; // shesh er digit
            sum += Math.pow(digit, digits);
            num = num / 10; // shesh er digit bad
        }

        // jogfol ashol number er soman hole Armstrong
        if (sum == original) {
            System.out.println(original + " is an Armstrong Number");
        } else {
            System.out.println(original + " is not an Armstrong Number");
        }
        sc.close();
    }
}


5. Strong Number

import java.util.Scanner;

public class StrongNumber {
    // ei method diye ekta number er factorial ber kora hoy
    public static int factorial(int n) {
        int fact = 1;
        for (int i = 1; i <= n; i++) {
            fact *= i; // 1 theke n porjonto gun korchi
        }
        return fact;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int num = sc.nextInt();
        int original = num; // ashol number er copy
        int sum = 0;

        // prottek digit er factorial ber kore jog korchi
        while (num > 0) {
            int digit = num % 10;
            sum += factorial(digit);
            num = num / 10;
        }

        // jogfol ar ashol number ek hole strong number
        if (sum == original) {
            System.out.println(original + " is a Strong Number");
        } else {
            System.out.println(original + " is not a Strong Number");
        }
        sc.close();
    }
}

9. Perfect Number

import java.util.Scanner;

public class PerfectNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int num = sc.nextInt();
        int sum = 0;

        // number ta bade baki shob utpadok (jegulo diye bhag jay) jog korchi
        for (int i = 1; i < num; i++) {
            if (num % i == 0) {
                sum += i;
            }
        }

        // utpadok gulor jogfol number er soman hole perfect number
        if (sum == num) {
            System.out.println(num + " is a Perfect Number");
        } else {
            System.out.println(num + " is not a Perfect Number");
        }
        sc.close();
    }
}

14. Pyramid

import java.util.Scanner;

public class Pyramid {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of rows: ");
        int rows = sc.nextInt();

        for (int i = 1; i <= rows; i++) {
            // age space dicchi jate star gulo majhkhane ashe
            for (int j = 1; j <= rows - i; j++) {
                System.out.print(" ");
            }
            // prottek row te star hobe 1, 3, 5, 7 ... eivabe
            for (int k = 1; k <= (2 * i - 1); k++) {
                System.out.print("*");
            }
            System.out.println();
        }
        sc.close();
    }
}


15. Diamond

import java.util.Scanner;

public class Diamond {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of rows (odd preferred): ");
        int n = sc.nextInt();

        // upor er ordhek, ekdom pyramid er moto
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int k = 1; k <= (2 * i - 1); k++) {
                System.out.print("*");
            }
            System.out.println();
        }

        // niche r ordhek, pyramid ta ulta kore boshalam
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int k = 1; k <= (2 * i - 1); k++) {
                System.out.print("*");
            }
            System.out.println();
        }
        sc.close();
    }
}

18. Simple Calculator (Switch-case)

import java.util.Scanner;

public class SimpleCalculator {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter first number: ");
        double num1 = sc.nextDouble();
        System.out.print("Enter second number: ");
        double num2 = sc.nextDouble();
        System.out.print("Enter operator (+, -, *, /): ");
        char op = sc.next().charAt(0); // shudhu prothom character ta nilam

        double result = 0;

        // switch diye operator onujayi kaj korachchi
        switch (op) {
            case '+':
                result = num1 + num2;
                break;
            case '-':
                result = num1 - num2;
                break;
            case '*':
                result = num1 * num2;
                break;
            case '/':
                // shunno diye bhag jay na, tai age check kore nilam
                if (num2 != 0) {
                    result = num1 / num2;
                } else {
                    System.out.println("Cannot divide by zero!");
                    sc.close();
                    return;
                }
                break;
            default:
                // user ulta palta kichu dile ekhane ashbe
                System.out.println("Invalid operator!");
                sc.close();
                return;
        }
        System.out.println("Result = " + result);
        sc.close();
    }
}

        int largest;

        // if-else diye ektar shathe arektar tulona korchi
        if (a >= b && a >= c) {
            largest = a;
        } else if (b >= a && b >= c) {
            largest = b;

        }

        System.out.println("Largest number is: " + largest);

    }
}

26. GCD

import java.util.Scanner;

public class GCD {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter first number: ");
        int a = sc.nextInt();
        System.out.print("Enter second number: ");
        int b = sc.nextInt();

        // Euclid er niyom: b shunno na hoa porjonto bhagshesh nite thaki
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        System.out.println("GCD = " + a); // shesh e a er moddhei uttor thake
        sc.close();
    }
}


27. LCM

import java.util.Scanner;

public class LCM {
    // LCM ber korte GCD lage, tai alada method banalam
    public static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter first number: ");
        int a = sc.nextInt();
        System.out.print("Enter second number: ");
        int b = sc.nextInt();

        // LCM = (a * b) / GCD
        int lcm = (a * b) / gcd(a, b);
        System.out.println("LCM = " + lcm);
        sc.close();
    }
}
